
#!/bin/bash

FILE="addressbook.txt"

# 1. Create Address Book
createAddressBook()
{
    if [ -f "$FILE" ]
    then
        echo "Address Book already exists."
    else
        touch "$FILE"
        echo "Address Book created successfully."
    fi
}

# 2. View Address Book
viewRecord()
{
    if [ ! -f "$FILE" ]
    then
        echo "Please create Address Book first."
    elif [ -s "$FILE" ]
    then
        echo
        echo "Address Book Records:"
        echo "---------------------"
        cat "$FILE"
    else
        echo "Address Book is empty."
    fi
}

# 3. Insert a Record
insertRecord()
{
    if [ ! -f "$FILE" ]
    then
        echo "Please create Address Book first."
        return
    fi

    echo "Enter Last Name:"
    read -r lastName

    echo "Enter First Name:"
    read -r firstName

    echo "Enter Email:"
    read -r email

    echo "Enter Phone Number:"
    read -r phone

    echo "$lastName | $firstName | $email | $phone" >> "$FILE"

    echo "Record inserted successfully."
}

# 4. Delete a Record
deleteRecord()
{
    if [ ! -f "$FILE" ]
    then
        echo "Address Book does not exist."
        return
    fi

    echo "Enter name or email to search:"
    read -r name

    grep -inF -- "$name" "$FILE"

    if [ $? -eq 0 ]
    then
        echo "Enter line number to delete:"
        read -r lineNumber

        if [[ "$lineNumber" =~ ^[0-9]+$ ]] &&
           [ "$lineNumber" -ge 1 ] &&
           [ "$lineNumber" -le "$(wc -l < "$FILE")" ]
        then
            if sed -n "${lineNumber}p" "$FILE" | grep -qiF -- "$name"
            then
                sed -i "${lineNumber}d" "$FILE"
                echo "Record deleted successfully."
            else
                echo "Selected record does not match."
            fi
        else
            echo "Invalid line number."
        fi
    else
        echo "No record found."
    fi
}

# 5. Modify a Record
modifyRecord()
{
    if [ ! -f "$FILE" ]
    then
        echo "Address Book does not exist."
        return
    fi

    echo "Enter name or email to search:"
    read -r name

    grep -inF -- "$name" "$FILE"

    if [ $? -eq 0 ]
    then
        echo "Enter line number to modify:"
        read -r lineNumber

        if [[ "$lineNumber" =~ ^[0-9]+$ ]] &&
           [ "$lineNumber" -ge 1 ] &&
           [ "$lineNumber" -le "$(wc -l < "$FILE")" ]
        then
            if sed -n "${lineNumber}p" "$FILE" | grep -qiF -- "$name"
            then
                echo "Enter new Last Name:"
                read -r lastName

                echo "Enter new First Name:"
                read -r firstName

                echo "Enter new Email:"
                read -r email

                echo "Enter new Phone Number:"
                read -r phone

                newRecord="$lastName | $firstName | $email | $phone"

                sed -i "${lineNumber}c\\$newRecord" "$FILE"

                echo "Record modified successfully."
            else
                echo "Selected record does not match."
            fi
        else
            echo "Invalid line number."
        fi
    else
        echo "No record found."
    fi
}

# Main Menu
while true
do
    echo
    echo "====== ADDRESS BOOK ======"
    echo "1. Create Address Book"
    echo "2. View Address Book"
    echo "3. Insert a Record"
    echo "4. Delete a Record"
    echo "5. Modify a Record"
    echo "6. Exit"
    echo "=========================="

    echo "Enter your choice:"
    read -r choice

    case $choice in
        1) createAddressBook ;;
        2) viewRecord ;;
        3) insertRecord ;;
        4) deleteRecord ;;
        5) modifyRecord ;;
        6) echo "Thank you!"
           exit 0 ;;
        *) echo "Invalid choice. Please try again." ;;
    esac
done
